# Antidote Plugin List
# This file defines the zsh plugins to be loaded by Antidote

# Prompt (Pure theme - same as Prezto)
sindresorhus/pure

# Git utilities (Oh My Zsh plugin)
# Load git lib functions first (provides git_current_branch, etc.)
ohmyzsh/ohmyzsh path:lib/git.zsh
ohmyzsh/ohmyzsh path:plugins/git

# Docker completion and utilities
ohmyzsh/ohmyzsh path:plugins/docker
ohmyzsh/ohmyzsh path:plugins/docker-compose

# SSH Agent management (needs_ssh_agent is defined in zshrc)
ohmyzsh/ohmyzsh path:plugins/ssh-agent conditional:needs_ssh_agent

# Additional completions
zsh-users/zsh-completions

# Syntax highlighting (must be loaded near the end)
zsh-users/zsh-syntax-highlighting

# Command suggestions from history
# zsh-users/zsh-autosuggestions

# History substring search (arrow keys)
zsh-users/zsh-history-substring-search
